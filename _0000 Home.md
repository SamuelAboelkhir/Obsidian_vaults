# Class
```plantuml
@startuml
class User {
  - id: string
  - email: string
  - password: string
  + login(): boolean
  + logout(): void
  + updateProfile(data: UserData): void
}

class UserService {
  - userRepository: UserRepository
  + createUser(userData: UserData): User
  + findById(id: string): User
  + deleteUser(id: string): void
}

class UserRepository {
  + save(user: User): void
  + findById(id: string): User
  + delete(id: string): void
}

User ||--o{ UserService : manages
UserService ||--|| UserRepository : uses
@enduml
```
# Sequence
```plantuml
@startuml
actor User
participant "Frontend" as FE
participant "AuthService" as Auth
participant "UserService" as US
participant "Database" as DB

User -> FE: login(email, password)
FE -> Auth: authenticate(email, password)
Auth -> US: findUserByEmail(email)
US -> DB: SELECT * FROM users
DB -> US: userData
US -> Auth: user
Auth -> Auth: validatePassword()
Auth -> FE: authToken
FE -> User: loginSuccess
@enduml
```
