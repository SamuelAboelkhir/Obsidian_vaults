# Class
```plantuml
@startuml
class User {
  - id: string
  - email: string
  - password: string
  + login(): boolean
  + logout(): void
  + updateProfile(data: UserData): void
}

class UserService {
  - userRepository: UserRepository
  + createUser(userData: UserData): User
  + findById(id: string): User
  + deleteUser(id: string): void
}

class UserRepository {
  + save(user: User): void
  + findById(id: string): User
  + delete(id: string): void
}

User ||--o{ UserService : manages
UserService ||--|| UserRepository : uses
@enduml
```

```plantuml
participant "UserService" as US

User -> FE: login(email, password)
FE -> Auth: authenticate(email, password)
Auth -> US: findUserByEmail(email)
US -> DB: SELECT * FROM users
DB -> US: userData
US -> Auth: user
Auth -> Auth: validatePassword()
Auth -> FE: authToken
FE -> User: loginSuccess
@enduml
```

```plantuml
@startuml
actor "Probill"
package "Client" {
	[Bill Of Lading]
	[Stakeholder]
	[Recepient]
	[EDI]
	[Order]
	[Contact]
}
package "Cargo" {
	[Shipment]
	[Item]
}
package "TMS" {
	[Trailer]
	[Dolly]
	[FleetLink]
	[Truck]
	[ISAAC]
	[GPS Location]
	[Accessory]
	[Trip]
	[External Trip]
	[Internal Trip]
}
package "WMS" {
	[Terminal]
	[Garage]
	[Yard]
	[Dock]
	[Warehouse]
	[Shunt]
}
package "Finance" {
	[Invoice]
	[Bank]
	[Overcharge]
}

[Stakeholder] --> [Bill Of Lading]: Generate BoL
[Bill Of Lading] --> [Truck]: To VA
[Shipment] --> [Truck]: To VA
[Truck] --> [WMS]: Back to warehouse
[WMS] -> Probill: Generate Probill
Probill --> [Recepient]: To recepient
[Recepient] --> Probill: Recepient signs
Probill --> [Finance]: Signed Probill to Finance
[Finance] --> [Invoice]: Generate invoice
[Invoice] --> [Stakeholder]: Bill Client
@enduml
```
```plantuml
@startuml
actor Client
participant "Truck" as TR
participant "Warehouse" as WH
participant "Data entry" as DE
participant "AS400" as A4
participant "Probill" as PR
participant "Recepient" as REC
participant "Finance" as FN

Client -> TR: BoL + Cargo
TR -> WH: Back to warehouse
WH -> DE: BoL delivered to operators
DE -> A4: BoL is scanned with librex onto AS400
A4 -> PR: Probill is printed
PR -> TR: Probil and cargo are loaded onto another truck
TR -> REC: Cargo delivered to recepient
REC -> PR: Probill signed
PR -> FN: Signed probill used to generate invoice
FN -> Client: Client billed
@enduml
```
